### 2. Context Carrier Protocol
```typescript
class ContextCarrier {
    private context: Map<string, any> = new Map();

    set(key: string, value: any): void {
        this.context.set(key, {
            value,
            timestamp: Date.now(),
            type: typeof value
        });
    }

    get(key: string): any {
        const item = this.context.get(key);
        if (!item) return null;

        // Check for stale context (older than 30 minutes)
        if (Date.now() - item.timestamp > 30 * 60 * 1000) {
            this.context.delete(key);
            return null;
        }

        return item.value;
    }

    // Auto-persist critical context
    persistCritical(keys: string[]): void {
        keys.forEach(key => {
            const value = this.get(key);
            if (value) {
                // Save to temporary storage
                this.persistToStorage(key, value);
            }
        });
    }
}
```

## ✅ OUTPUT VALIDATION SYSTEM

### 1. Validation Rules Engine
```typescript
class OutputValidator {
    private rules: ValidationRule[] = [
        new LanguageRule(),           // Indonesian communication
        new ToolUsageRule(),          // MCP tools properly used
        new CodeStyleRule(),          // Project conventions
        new SecurityRule(),           // Security best practices
        new PerformanceRule(),        // Performance considerations
        new AccessibilityRule(),      // WCAG compliance
        new CompletenessRule(),       // Holy Trinity check
        new DomainRule(),             // Within knowledge boundaries
    ];

    validate(response: AIResponse): ValidationResult {
        const results = this.rules.map(rule => rule.validate(response));

        return {
            passed: results.every(r => r.passed),
            score: this.calculateScore(results),
            violations: results.filter(r => !r.passed),
            suggestions: this.generateSuggestions(results)
        };
    }
}

class LanguageRule implements ValidationRule {
    validate(response: AIResponse): RuleResult {
        const hasIndonesian = /\\b(tambah|buat|edit|hapus|perbaiki|cek|analisa)\\b/i.test(response.implementation?.description || '');
        const codeInEnglish = !(/\\b(membuat|menambah|mengedit|menghapus)\\b/i.test(response.implementation?.code || ''));

        return {
            passed: hasIndonesian && codeInEnglish,
            message: hasIndonesian && codeInEnglish ? 'OK' : 'Use Indonesian for communication, English for code'
        };
    }
}
```

### 2. Quality Score Calculation
```typescript
interface QualityMetrics {
    completeness: number;      // Holy Trinity fulfillment
    correctness: number;       // Technical accuracy
    consistency: number;       // Follows project patterns
    performance: number;       // Performance considerations
    security: number;          // Security best practices
    maintainability: number;   // Code quality and documentation
}

class QualityScorer {
    calculate(response: AIResponse): QualityMetrics {
        return {
            completeness: this.scoreCompleteness(response),
            correctness: this.scoreCorrectness(response),
            consistency: this.scoreConsistency(response),
            performance: this.scorePerformance(response),
            security: this.scoreSecurity(response),
            maintainability: this.scoreMaintainability(response)
        };
    }

    private scoreCompleteness(response: AIResponse): number {
        const hasBackend = response.implementation?.backend_logic ? 1 : 0;
        const hasFrontend = response.implementation?.ui_specification ? 1 : 0;
        const hasDatabase = response.analysis?.database_schema ? 1 : 0;

        return (hasBackend + hasFrontend + hasDatabase) / 3 * 100;
    }
}
```

## 🎯 PROMPT OPTIMIZATION ENGINE

### 1. Prompt Improvement Algorithm
```typescript
class PromptOptimizer {
    optimize(originalPrompt: string, context: TaskContext): OptimizedPrompt {
        const improvements = [];

        // 1. Add task type specification
        if (!this.hasTaskType(originalPrompt)) {
            improvements.push('Add specific task type (feature/bugfix/optimization)');
        }

        // 2. Add scope clarification
        if (!this.hasScope(originalPrompt)) {
            improvements.push('Specify scope (frontend/backend/fullstack)');
        }

        // 3. Add requirements array
        if (!this.hasRequirements(originalPrompt)) {
            improvements.push('List specific requirements as array');
        }

        // 4. Add constraints
        if (!this.hasConstraints(originalPrompt)) {
            improvements.push('Define constraints and limitations');
        }

        // 5. Add priority level
        if (!this.hasPriority(originalPrompt)) {
            improvements.push('Set priority level');
        }

        return {
            original: originalPrompt,
            optimized: this.applyImprovements(originalPrompt, improvements),
            improvements: improvements,
            improvement_score: improvements.length,
            confidence_boost: this.calculateConfidenceBoost(improvements)
        };
    }

    private applyImprovements(prompt: string, improvements: string[]): string {
        // Template-based prompt reconstruction
        const template = `
Task Type: [feature/bugfix/optimization/debug]
Priority: [low/medium/high/critical]
Scope: [frontend/backend/fullstack/database]
Target Area: [specific component/module]
Description: ${prompt}

Requirements:
- [requirement 1]
- [requirement 2]
- [requirement 3]

Constraints:
- [constraint 1]
- [constraint 2]

Context Level: [minimal/standard/comprehensive]
Follow AI Rules: yes
Output Format: [standard/detailed/minimal]
        `.trim();

        return template;
    }
}
```
